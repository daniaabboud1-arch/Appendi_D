\\Fidelity and Fawzi–Renner Bound
as Functions of t


import numpy as np
import matplotlib.pyplot as plt

p = 0.1
t_values = np.linspace(0, 0.02, 51)

def Pdist(p,t):
    Pxy = np.array([
        [(1-p)**2+t, p*(1-p)-t],
        [p*(1-p)-t, p**2+t]
    ])
    P = np.zeros((2,2,2))
    for b in range(2):
        for x in range(2):
            for y in range(2):
                P[b,b^y,b^x] += 0.5*Pxy[x,y]
    return P

def CMI(P):
    PB,PBR,PBC = P.sum((1,2)),P.sum(1),P.sum(2)
    return sum(
        P[b,c,r]*np.log2(
            P[b,c,r]*PB[b]/(PBR[b,r]*PBC[b,c])
        )
        for b in range(2)
        for c in range(2)
        for r in range(2)
        if P[b,c,r]>0
    )

Factual = [ ]
FRbound = [ ]

for t in t_values:
    P = Pdist(p,t)
    I = CMI(P)
    PB,PBR,PBC = P.sum((1,2)),P.sum(1),P.sum(2)

    Q = PBR[:,None,:]*PBC[:,:,None]/PB[:,None,None]

    Factual.append(np.sum(np.sqrt(P*Q)))
    FRbound.append(2**(-I/2))

Factual,FRbound = np.array(Factual),np.array(FRbound)

print(f"Actual fidelity = {Factual[-1]:.6f}")
print(f"FR bound = {FRbound[-1]:.6f}")

plt.figure(figsize=(7,5))

plt.plot(
    t_values,
    Factual,
    label="Actual Fidelity"
)

plt.plot(
    t_values,
    FRbound,
    label="Fawzi-Renner Bound"
)

plt.scatter(
    0.02,
    Factual[-1]
)

plt.xlabel(r"$t$")
plt.ylabel("Fidelity")
plt.legend()
plt.grid()
plt.tight_layout()
plt.show()
